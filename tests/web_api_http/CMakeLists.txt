cmake_minimum_required(VERSION 3.20.0)

set(APP_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
list(APPEND ZEPHYR_EXTRA_MODULES
    ${APP_ROOT}/modules/api-validation
    ${APP_ROOT}/modules/job-manager
    ${APP_ROOT}/modules/web-auth
    ${APP_ROOT}/modules/web-api
    ${APP_ROOT}/modules/matter-service
    ${APP_ROOT}/modules/web-assets
    ${APP_ROOT}/modules/device-config-store
    ${APP_ROOT}/modules/network-manager
    ${APP_ROOT}/modules/log-store
    ${APP_ROOT}/modules/coprocessor-manager
    ${APP_ROOT}/modules/firmware-store
)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(web_api_http_test)

# The assets table is generated by the same script the firmware uses, from a
# small fixture dist, so the suite serves what the generator really produces.
set(gen_dir ${CMAKE_BINARY_DIR}/generated/test)
add_custom_command(
    OUTPUT ${gen_dir}/web_assets_data.c
    COMMAND ${PYTHON_EXECUTABLE} ${APP_ROOT}/tools/web-assets/gen_web_assets.py
            --dist ${CMAKE_CURRENT_SOURCE_DIR}/fixture-dist
            --out-c ${gen_dir}/web_assets_data.c
    DEPENDS ${APP_ROOT}/tools/web-assets/gen_web_assets.py
            ${CMAKE_CURRENT_SOURCE_DIR}/fixture-dist/index.html
            ${CMAKE_CURRENT_SOURCE_DIR}/fixture-dist/assets/app-AbCdEf12.js
    VERBATIM
)

set(V1_DIR ${APP_ROOT}/src/web/api/v1)
target_include_directories(app PRIVATE ${V1_DIR})
target_sources(app PRIVATE
    src/main.c
    ${gen_dir}/web_assets_data.c
    ${V1_DIR}/router.c
    ${V1_DIR}/auth.c
    ${V1_DIR}/system.c
    ${V1_DIR}/matter.c
    ${V1_DIR}/network.c
    ${V1_DIR}/logs.c
    ${V1_DIR}/coprocessor.c
    ${V1_DIR}/firmware.c
    ${V1_DIR}/system_update.c
)

zephyr_linker_sources(SECTIONS sections-rom.ld)
zephyr_iterable_section(NAME http_resource_desc_test_service KVMA RAM_REGION GROUP RODATA_REGION)
